        <p>Каждый раз писать четыре команды перемещения — лень. Поэтому программисты упаковывают это в одну удобную микро-функцию:</p>
        
        <pre><code><span class="keyword">def</span> <span class="func">p</span>(x, y):
    <span class="func">penup</span>()       <span class="comment"># 1. Подняли хвост (не рисуем)</span>
    <span class="func">setpos</span>(x, y)  <span class="comment"># 2. Телепортировались в точку (x, y)</span>
    <span class="func">pendown</span>()     <span class="comment"># 3. Опустили хвост (готовы рисовать)</span>
    <span class="func">seth</span>(<span class="number">0</span>)       <span class="comment"># 4. Повернулись строго на восток (0 градусов)</span></code></pre>
        
        <div class="analogy-box">
          <strong>Зачем <code>seth(0)</code>?</strong> Чтобы сбросить угол поворота Черепахи. Иначе после предыдущего рисунка её голова будет смотреть куда попало!

            <li><code>circle(50, 360, 4)</code> — <strong>квадрат!</strong> Параметр <code>steps=4</code> говорит Черепахе сделать 4 шага-угла.</li>

<span class="comment"># ?? Розан: лепестки-квадраты</span>

    /* ========================================================

    const canvas = document.getElementById('turtleCanvas');
    const ctx = canvas.getContext('2d');

    function clearCanvas() {
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      ctx.fillStyle = "#020617";
      ctx.fillRect(0, 0, canvas.width, canvas.height);
    }

    function drawGridCircles() {
      clearCanvas();
      const r = 8;
      const step = r * 2;
      for (let y = 30; y <= canvas.height - 30; y += step) {
        for (let x = 30; x <= canvas.width - 30; x += step) {
          ctx.beginPath();
          ctx.arc(x, y, r, 0, Math.PI * 2);
          ctx.strokeStyle = "#38bdf8";
          ctx.fillStyle = "rgba(6, 182, 212, 0.2)";
          ctx.fill();
          ctx.stroke();
        }
      }
    }

    function drawGeorgin(n = 12, r = 50) {
      clearCanvas();
      const cx = canvas.width / 2;
      const cy = canvas.height / 2;
      
      for (let i = 0; i < n; i++) {
        const angle = (i * 2 * Math.PI) / n;
        // Круг, проходящий через центр
        const ox = cx + r * Math.cos(angle);
        const oy = cy + r * Math.sin(angle);
        
        ctx.beginPath();
        ctx.arc(ox, oy, r, 0, Math.PI * 2);
        ctx.strokeStyle = "#f472b6";
        ctx.lineWidth = 1.5;
        ctx.fillStyle = "rgba(244, 114, 182, 0.08)";
        ctx.fill();
        ctx.stroke();
      }
    }

    function drawRomashka(n = 8, r = 60) {
      clearCanvas();
      const cx = canvas.width / 2;
      const cy = canvas.height / 2;

      for (let i = 0; i < n; i++) {
        const angle = (i * 2 * Math.PI) / n;
        ctx.save();
        ctx.translate(cx, cy);
        ctx.rotate(angle);

        ctx.beginPath();
        ctx.ellipse(0, -r/2, r/4, r/2, 0, 0, Math.PI * 2);
        ctx.fillStyle = "rgba(250, 204, 21, 0.25)";
        ctx.strokeStyle = "#facc15";
        ctx.lineWidth = 1.5;
        ctx.fill();
        ctx.stroke();
        ctx.restore();
      }

      // Центр цветка
      ctx.beginPath();
      ctx.arc(cx, cy, 14, 0, Math.PI * 2);
      ctx.fillStyle = "#fb923c";
      ctx.fill();
      ctx.strokeStyle = "#fff";
      ctx.stroke();
    }

    function drawRozan(n = 8, size = 60) {
      clearCanvas();
      const cx = canvas.width / 2;
      const cy = canvas.height / 2;

      for (let i = 0; i < n; i++) {
        const angle = (i * 2 * Math.PI) / n;
        ctx.save();
        ctx.translate(cx, cy);
        ctx.rotate(angle);

        ctx.beginPath();
        ctx.rect(0, 0, size, size);
        ctx.strokeStyle = "#a855f7";
        ctx.fillStyle = "rgba(168, 85, 247, 0.15)";
        ctx.lineWidth = 1.5;
        ctx.fill();
        ctx.stroke();
        ctx.restore();
      }
    }

    function drawRandomStars() {
      clearCanvas();
      for (let i = 0; i < 40; i++) {
        const x = Math.random() * canvas.width;
        const y = Math.random() * canvas.height;
        const r = 2 + Math.random() * 8;
        const colors = ["#38bdf8", "#facc15", "#f472b6", "#4ade80", "#a78bfa"];
        const col = colors[Math.floor(Math.random() * colors.length)];

        ctx.beginPath();
        ctx.arc(x, y, r, 0, Math.PI * 2);
        ctx.fillStyle = col;
        ctx.shadowColor = col;
        ctx.shadowBlur = 8;
        ctx.fill();
      }
      ctx.shadowBlur = 0;
    }

      else if (type === 'rozan') drawRozan();
      else if (type === 'grid') drawGridCircles();
      else if (type === 'random_stars') drawRandomStars();
    }

    let currentDiff = 'std';
    let userAnswers = {};

    function setDifficulty(diff) {
      currentDiff = diff;
      document.getElementById('btn-std').classList.toggle('active', diff === 'std');
      document.getElementById('btn-pro').classList.toggle('active', diff === 'pro');

      renderPractice();
      renderQuiz();
    }

    function cleanString(str) {
      return str.toLowerCase().replace(/\s+/g, '').replace(/[();]/g, '');
    }

    function renderPractice() {
      const container = document.getElementById('practice-container');
      const tasks = tasksData[currentDiff];

      container.innerHTML = tasks.map((t, idx) => `
        <div class="practice-item" id="task-card-${t.id}">
          <div class="practice-header">
            <span style="color: #38bdf8;">Задание ${idx + 1} • ${t.type}</span>
          </div>
          <div class="task-desc">${t.text}</div>
          <div class="interactive-row">
            <input type="text" id="input-${t.id}" placeholder="${t.placeholder}">
            <button class="btn btn-primary" onclick="checkTask(${t.id})">Проверить</button>
            <button class="btn btn-secondary" onclick="toggleSolution(${t.id})">Объяснение</button>
            <span id="feedback-${t.id}"></span>
          </div>
          <div class="explanation" id="expl-${t.id}">${t.solution}</div>
        </div>
      `).join('');
    }

    function checkTask(id) {
      const task = tasksData[currentDiff].find(t => t.id === id);
      const val = cleanString(document.getElementById(`input-${id}`).value);
      const feedback = document.getElementById(`feedback-${id}`);

      const isCorrect = task.validAnswers.some(ans => cleanString(ans) === val);

      if (isCorrect) {
        feedback.innerHTML = `<span class="feedback-icon ok">? Точно в яблочко!</span>`;
      } else {
        feedback.innerHTML = `<span class="feedback-icon bad">? Не совсем. Нажми «Объяснение»</span>`;
      }
    }

    function toggleSolution(id) {
      const el = document.getElementById(`expl-${id}`);
      el.style.display = el.style.display === 'block' ? 'none' : 'block';
    }

    function renderQuiz() {
      const container = document.getElementById('quiz-container');
      const questions = quizData[currentDiff];
      userAnswers = {};
      document.getElementById('score-banner').style.display = 'none';

      container.innerHTML = questions.map((q, qIdx) => `
        <div class="test-card" id="q-card-${qIdx}">
          <div style="font-weight: 700; margin-bottom: 12px; font-size: 1.05rem;">
            Вопрос ${qIdx + 1}: ${q.q}
          </div>
          <div class="options-list">
            ${q.options.map((opt, oIdx) => `
              <button class="option-btn" id="opt-${qIdx}-${oIdx}" onclick="selectOption(${qIdx}, ${oIdx})">
                ${opt}
              </button>
            `).join('')}
          </div>
          <div class="explanation" id="quiz-expl-${qIdx}" style="margin-top: 10px;">${q.hint}</div>
        </div>
      `).join('') + `
        <div style="text-align: center; margin-top: 20px;">
          <button class="btn btn-primary" style="padding: 14px 30px; font-size: 1rem;" onclick="finishQuiz()">Завершить тест и узнать балл</button>
        </div>
      `;
    }

    function selectOption(qIdx, oIdx) {
      userAnswers[qIdx] = oIdx;
      const count = quizData[currentDiff][qIdx].options.length;
      for (let i = 0; i < count; i++) {
        document.getElementById(`opt-${qIdx}-${i}`).classList.remove('selected');
      }
      document.getElementById(`opt-${qIdx}-${oIdx}`).classList.add('selected');
    }

    function finishQuiz() {
      const questions = quizData[currentDiff];
      let score = 0;

      questions.forEach((q, qIdx) => {
        const userChoice = userAnswers[qIdx];
        const expl = document.getElementById(`quiz-expl-${qIdx}`);
        expl.style.display = 'block';

        if (userChoice !== undefined) {
          const btn = document.getElementById(`opt-${qIdx}-${userChoice}`);
          if (userChoice === q.correct) {
            btn.classList.add('correct');
            score++;
          } else {
            btn.classList.add('wrong');
            document.getElementById(`opt-${qIdx}-${q.correct}`).classList.add('correct');
          }
        } else {
          document.getElementById(`opt-${qIdx}-${q.correct}`).classList.add('correct');
        }
      });

      const banner = document.getElementById('score-banner');
      const title = document.getElementById('score-title');
      const desc = document.getElementById('score-desc');

      banner.style.display = 'block';
      title.innerText = `Правильно: ${score} из ${questions.length}`;

      if (score === questions.length) {
        desc.innerText = "?? Идеально! Ты настоящий гуру алгоритмической графики!";
      } else if (score >= questions.length / 2) {
        desc.innerText = "?? Хороший результат! Разбери подсказки к тем вопросам, где ошибся.";
      } else {
        desc.innerText = "?? Попробуй перечитать теорию и нажать кнопки в симуляторе, а затем повторить!";
      }

      banner.scrollIntoView({ behavior: 'smooth' });
    }

    function restartQuiz() {
      renderQuiz();
